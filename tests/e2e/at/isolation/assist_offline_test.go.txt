func TestENR02_CloneRuntimeReadiness(t *testing.T) {
	dir := t.TempDir()
	write := func(name, body string, mode os.FileMode) {
		t.Helper()
		if err := os.WriteFile(filepath.Join(dir, name), []byte(body), mode); err != nil {
			t.Fatal(err)
		}
	}
	// TEST ONLY: every docker invocation terminates in this local fixture.
	// Unknown arguments fail; no real container or network operation is called.
	write("docker", `#!/usr/bin/python3
import json, os, sys
from pathlib import Path
root=Path(os.environ['FVPN_TEST_RUNTIME_READS'])
args=sys.argv[1:]
with (root/'calls.jsonl').open('a') as f:f.write(json.dumps(args)+'\n')
if args[0]=='inspect':
 print(json.dumps({'id':'TEST ONLY container','started_at':'2026-01-01T00:00:00Z','running':True}))
elif args[0]=='cp':pass
elif args[0]=='exec' and args[2:]==['stat','-f','-c','%T','/tmp']:print('ext2/ext3')
elif args[0]=='exec' and args[2:]==['/tmp/isolation-probe','runtime-evidence']:
 role='clone' if args[1].endswith('-clone') else 'source'
 count=root/(role+'.count')
 n=int(count.read_text()) if count.exists() else 0
 count.write_text(str(n+1))
 print((root/(role+str(min(n,1))+'.json')).read_text())
else:raise SystemExit('unexpected fake docker command')
`, 0o755)
	snapshot := func(marker string, coherent bool) string {
		return fmt.Sprintf(`{"marker":%q,"read_start":%q,"read_end":%q,"process_coherent":%t,"process_before":{"status":"available","role":"fvpn-agent","pid":42,"start_ticks":"100"},"process_after":{"status":"available","role":"fvpn-agent","pid":42,"start_ticks":"100"},"agent_pid":{"status":"available","text":"42\n"},"agent_process_id":{"status":"available","text":"00000000-0000-4000-8000-000000000001\n"}}`, marker, time.Now().UTC().Format(time.RFC3339Nano), time.Now().UTC().Format(time.RFC3339Nano), coherent)
	}
	first, ready := snapshot("TEST ONLY first incoherent read", false), snapshot("TEST ONLY exact ready read", true)
	write("source0.json", first, 0o600)
	write("source1.json", ready, 0o600)
	write("clone0.json", snapshot("TEST ONLY clone read", true), 0o600)
	write("clone1.json", snapshot("TEST ONLY later clone read", true), 0o600)
	t.Setenv("PATH", dir+string(os.PathListSeparator)+os.Getenv("PATH"))
	t.Setenv("FVPN_TEST_RUNTIME_READS", dir)
	ctx, cancel := context.WithCancel(t.Context())
	defer cancel()
	x := &isolationRun{t: t, ctx: ctx, dir: dir, binary: "TEST ONLY probe",
		r:       &runner{lab: &lab{prefix: "fvpn-w4-isolation", node: "fvpn-w4-isolation-node"}},
		records: map[string]*iso.Record{"AT13": {RunID: "TEST ONLY run", Build: strings.Repeat("a", 40)}},
	}
	if !x.retainRuntime("after-clone-creation") || len(x.runtimeEvidence) != 2 {
		t.Fatalf("ready source/clone phase not retained: %v", x.runtimeEvidence)
	}
	if string(x.runtimeEvidence[0]["snapshot"].(json.RawMessage)) != ready+"\n" {
		t.Fatal("successful source read was replaced or refetched")
	}
	var saved struct {
		Snapshots []struct {
			Role, Phase, Status string
			Readiness           struct {
				Status   string
				Attempts []struct {
					Snapshot  json.RawMessage
					ReadStart time.Time `json:"read_start"`
					ReadEnd   time.Time `json:"read_end"`
				}
			}
		}
	}
	raw, err := os.ReadFile(filepath.Join(dir, "clone-runtime.json"))
	if err != nil || json.Unmarshal(raw, &saved) != nil || len(saved.Snapshots) != 2 {
		t.Fatalf("phase artifact not written: %s %v", raw, err)
	}
	readiness := saved.Snapshots[0].Readiness
	var original, retained any
	_ = json.Unmarshal([]byte(first), &original)
	if len(readiness.Attempts) != 1 || readiness.Status != "ready" {
		t.Fatalf("first failed source read missing: %s", raw)
	}
	_ = json.Unmarshal(readiness.Attempts[0].Snapshot, &retained)
	if !reflect.DeepEqual(original, retained) || readiness.Attempts[0].ReadStart.IsZero() || readiness.Attempts[0].ReadEnd.Before(readiness.Attempts[0].ReadStart) || saved.Snapshots[0].Role != "source" || saved.Snapshots[0].Phase != "after-clone-creation" {
		t.Fatal("readiness attempt replaced the unique source candidate")
	}
	// These independent closure reads must still run after the main context ends.
	cancel()
	x.retainRuntime("after-clone-observation")
	if len(x.runtimeEvidence) != 4 || x.runtimeEvidence[2]["status"] != "available" || x.runtimeEvidence[3]["status"] != "available" {
		t.Fatal("cancellation suppressed independent post-observation evidence")
	}
	for role, want := range map[string]string{"source": "3", "clone": "2"} {
		n, err := os.ReadFile(filepath.Join(dir, role+".count"))
		if err != nil || string(n) != want {
			t.Fatalf("unexpected %s sample count: %s %v", role, n, err)
		}
	}
	failed := *x
	failed.runtimeEvidence = nil
	failed.dir = filepath.Join(dir, "canceled")
	if failed.retainRuntime("after-clone-creation") || len(failed.runtimeEvidence) != 2 || failed.runtimeEvidence[0]["status"] != "unavailable" {
		t.Fatal("canceled initial source read qualified as a bracket")
	}
}

func TestAPI01_IsolationAssist(t *testing.T) {
	for _, tc := range []struct {
		name, code, outcome string
		status, writes      int
		already, accepted   bool
	}{
		{"accepted", "", "accepted", 202, 1, false, true},
		{"conflict_then_accepted", "RESOURCE_VERSION_CONFLICT", "accepted", 412, 2, false, true},
		{"non_conflict", "INTERNAL_ERROR", "non_conflict_response", 500, 1, false, false},
		{"wrong_412_code", "PRECONDITION_FAILED", "non_conflict_response", 412, 1, false, false},
		{"already_intended", "RESOURCE_VERSION_CONFLICT", "already_intended", 412, 1, true, true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			const path = "/api/v1/admin/nodes/test-only"
			const capacity = `{"capacity":{"down_bps":208000000,"member_down_bps_cap":100000000,"member_up_bps_cap":100000000,"up_bps":200000000}}`
			var mu sync.Mutex
			var methods, bodies, keys, tags []string
			reads, writes := 0, 0
			server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				mu.Lock()
				defer mu.Unlock()
				methods = append(methods, r.Method)
				if r.URL.Path != path {
					t.Errorf("unexpected path %q", r.URL.Path)
				}
				if r.Method == "GET" {
					reads++
					w.Header().Set("ETag", fmt.Sprintf(`"%d"`, reads))
					if reads > 1 && tc.already {
						_, _ = io.WriteString(w, capacity)
					} else {
						// Partial desired state must not satisfy nodeFieldsMatch.
						_, _ = io.WriteString(w, `{"capacity":{"down_bps":208000000}}`)
					}
					return
				}
				if r.Method != "PATCH" {
					t.Errorf("unexpected method %s", r.Method)
				}
				writes++
				raw, _ := io.ReadAll(r.Body)
				bodies = append(bodies, string(raw))
				keys = append(keys, r.Header.Get("Idempotency-Key"))
				tags = append(tags, r.Header.Get("If-Match"))
				if writes == 1 && tc.status != 202 {
					w.WriteHeader(tc.status)
					_, _ = fmt.Fprintf(w, `{"error":{"code":%q,"details":{"current_resource_version":"2"}}}`, tc.code)
					return
				}
				w.WriteHeader(202)
				_, _ = io.WriteString(w, `{"operation_id":"TEST ONLY"}`)
			}))
			defer server.Close()
			restored := 0
			evidence, err := isolationAssist(t.Context(), &session{http: server.Client()}, server.URL+path, func() ([]byte, error) {
				restored++
				return []byte("TEST ONLY restore stub"), nil
			})
			if err != nil {
				t.Fatal(err)
			}
			raw, err := json.Marshal(evidence)
			if err != nil {
				t.Fatal(err)
			}
			var got struct {
				At          time.Time       `json:"at"`
				CompletedAt time.Time       `json:"completed_at"`
				Action      string          `json:"action"`
				APIOK       bool            `json:"api_ok"`
				Assisted    bool            `json:"assisted"`
				RestoreOK   bool            `json:"restore_ok"`
				RestoreOut  string          `json:"restore_output"`
				Status      int             `json:"http_status"`
				Response    json.RawMessage `json:"response"`
				Receipt     nodeWriteTrace  `json:"node_write"`
			}
			if err := json.Unmarshal(raw, &got); err != nil {
				t.Fatal(err)
			}
			mu.Lock()
			defer mu.Unlock()
			if restored != 1 || !got.RestoreOK || got.RestoreOut != "TEST ONLY restore stub" || !got.Assisted || got.APIOK != tc.accepted {
				t.Fatalf("assist classification/acceptance/restore: %s", raw)
			}
			if got.Action != "Extra clean capacity generation (208000000) after recovery stalled; restore saved lab member addresses/routes again. No row or observation edits." || got.At.IsZero() || got.CompletedAt.Before(got.At) {
				t.Fatalf("changed action/time evidence: %s", raw)
			}
			trace := got.Receipt
			if trace.Outcome != tc.outcome || trace.MaxAttempts != 3 || trace.LimitSeconds != 10 || len(trace.Attempts) != tc.writes || writes != tc.writes {
				t.Fatalf("missing/bad receipt: %s", raw)
			}
			wantTraceStatus := StatusBlocked
			if tc.accepted {
				wantTraceStatus = StatusPass
			}
			if trace.Status != wantTraceStatus {
				t.Fatalf("receipt status: %s", raw)
			}
			wantMethods := []string{"GET"}
			previous := trace.StartedAt
			if previous.Before(got.At) {
				t.Fatal("logical command start moved after helper start")
			}
			for i, attempt := range trace.Attempts {
				wantMethods = append(wantMethods, "PATCH")
				if attempt.Method != "PATCH" || attempt.Path != path || attempt.IfMatch != tags[i] || tags[i] != fmt.Sprintf(`"%d"`, i+1) || attempt.IdempotencyKey == "" || attempt.IdempotencyKey != keys[i] || keys[i] != keys[0] || bodies[i] != capacity || attempt.BodySHA256 != fmt.Sprintf("%x", sha256.Sum256([]byte(bodies[i]))) {
					t.Fatalf("write/key/body/ETag evidence %d: %s", i, raw)
				}
				status := 202
				if i == 0 {
					status = tc.status
				}
				if attempt.Status != status || attempt.StartedAt.Before(previous) || attempt.FinishedAt.Before(attempt.StartedAt) {
					t.Fatalf("attempt status/time %d: %s", i, raw)
				}
				previous = attempt.FinishedAt
				if i == 0 && tc.code == "RESOURCE_VERSION_CONFLICT" {
					wantMethods = append(wantMethods, "GET")
					read := attempt.Reread
					if attempt.ErrorCode != tc.code || read == nil || read.Method != "GET" || read.Path != path || read.Status != 200 || read.ETag != `"2"` || read.Intended != tc.already || read.StartedAt.Before(previous) || read.FinishedAt.Before(read.StartedAt) {
						t.Fatalf("missing/bad reread: %s", raw)
					}
					previous = read.FinishedAt
				} else if attempt.Reread != nil {
					t.Fatalf("unexpected reread: %s", raw)
				}
			}
			if trace.FinishedAt.Before(previous) || got.CompletedAt.Before(trace.FinishedAt) || !reflect.DeepEqual(methods, wantMethods) || reads != len(wantMethods)-writes {
				t.Fatalf("request order/time: %v expected %v; %s", methods, wantMethods, raw)
			}
			if got.Status != trace.Attempts[len(trace.Attempts)-1].Status || !json.Valid(got.Response) {
				t.Fatalf("last response not retained: %s", raw)
			}
			t.Logf("TEST ONLY fake-server evidence: %s", raw)
		})
	}
}
