func TestIsolationAssembled(t *testing.T) {
	if os.Getenv("FVPN_E2E") != "1" {
		t.Skip("requires assigned devstack")
	}
	if os.Getenv("FVPN_E2E_PREFIX") != "w4-isolation" || os.Getenv("FVPN_E2E_PORT_BASE") != "36000" {
		t.Fatal("wrong assigned binding")
	}
	ctx, cancel := context.WithTimeout(t.Context(), 50*time.Minute)
	defer cancel()
	root, err := os.Getwd()
	if err != nil {
		t.Fatal(err)
	}
	root = filepath.Clean(filepath.Join(root, "../.."))
	dir := filepath.Join(root, "docs/e2e/at/isolation")
	if configured := os.Getenv("FVPN_AT_EVIDENCE_DIR"); configured != "" {
		if filepath.IsAbs(configured) {
			dir = configured
		} else {
			dir = filepath.Join(root, configured)
		}
	}
	source, _ := exec.Command("git", "rev-parse", "HEAD").Output()
	dirty, _ := exec.Command("git", "status", "--porcelain", "--", "apps", "node", "internal", "pkg", "tests", "deploy", "go.mod", "go.sum").Output()
	runID := os.Getenv("FVPN_AT_RUN_ID")
	if runID == "" {
		runID = uuid.NewString()
	}
	x := &isolationRun{t: t, ctx: ctx, dir: dir, records: map[string]*iso.Record{}}
	for _, id := range []string{"AT10", "AT11", "AT12", "AT13"} {
		x.records[id] = &iso.Record{ID: id, Title: "Assembled isolation " + id, Spec: []string{"§19 " + id}, Status: iso.StatusBlocked, RunID: runID, StartedAt: time.Now().UTC().Format(time.RFC3339), Build: strings.TrimSpace(string(source)), SourceDirty: len(dirty) > 0, Actual: "setup not completed", Commands: []string{"make -f mk/at-isolation.mk at-isolation"}}
		x.records[id].Expected = map[string]string{
			"AT10": "Protected host, intranet and metadata destinations stay denied during interface changes, with the same member's counted origin control.",
			"AT11": "IPv4 and IPv6 platform routes cannot bypass protected destinations; unavailable platforms remain blocked.",
			"AT12": "Same-member and cross-member peers remain isolated; publication groups withdraw excluded members; DNS and source identity exceptions require independent proof.",
			"AT13": "Disable, revoke and signed expiry withdraw forwarding; restart recovers safely and unmeasured fault windows remain blocked.",
		}[id]
	}
	defer x.finish()
	lab, err := openLab()
	if err != nil {
		x.problem("w3-devstack-e2e", err.Error())
		return
	}
	if lab.prefix != "fvpn-w4-isolation" {
		x.problem("acceptance", "unexpected stack prefix")
		return
	}
	defer lab.transport.CloseIdleConnections()
	x.r = &runner{t: t, lab: lab, dir: filepath.Join(dir, "support"), runID: runID, sourceCommit: strings.TrimSpace(string(source))}
	x.stack = &Stack{lab: lab}
	for _, s := range []string{"control", "worker", "node", "member", "netbird", "adguard", "gateway", "origin", "idp", "postgres"} {
		raw, e := exec.Command("docker", "inspect", "--format={{.Image}}", lab.prefix+"-"+s).Output()
		for _, rec := range x.records {
			if e == nil {
				rec.ImageDigests = append(rec.ImageDigests, s+" "+strings.TrimSpace(string(raw)))
			}
		}
	}
	x.admin, err = lab.login(ctx, "e2e-admin", "admin")
	if err != nil {
		x.problem("w3-devstack-e2e", "admin login failed")
		return
	}
	x.binary = filepath.Join(t.TempDir(), "isolation-probe")
	cmd := exec.CommandContext(ctx, "go", "build", "-o", x.binary, filepath.Join(root, "tests/e2e/at/isolation/probe_main.go"), filepath.Join(root, "tests/e2e/at/isolation/observer_main.go"))

		x.problem("acceptance", string(out))

func (x *isolationRun) problem(owner, detail string) {
	x.t.Logf("blocked: %s: %s", owner, detail)
	for _, rec := range x.records {
		rec.Actual = detail
		rec.Observations = append(rec.Observations, iso.Observation{Name: "owner", Detail: owner})
	}
}

func (x *isolationRun) finish() {
	x.stopConvergence()
	x.writeConvergence()
	x.archiveReceiver()
	if x.r != nil {
		raw, _ := exec.Command("docker", "logs", x.r.lab.node).CombinedOutput()
		_ = iso.WriteExtra(x.dir, "node-events.txt", iso.Redact(string(raw)))
	}
	x.reviewFollowup8()
	x.noteEvaluator()
	for _, id := range []string{"AT10", "AT11", "AT12", "AT13"} {
		rec := x.records[id]
		if x.r != nil {
			_, _, _, generation := x.r.desiredMemberView()
			rec.PolicyRevision = generation
			rec.NodeIncarnation = x.r.incID.String()
			rec.Observations = append(rec.Observations, iso.Observation{Name: "mechanism_control", Detail: "lab-" + id + ".json; separate mechanism proof, never substitutes for a counted product control"})
		}
		if len(rec.Checks) > 0 {
			rec.Status = iso.ChecksStatus(rec.Checks)
			rec.Actual = "See per-check positive controls, outcomes and owners."
		}
		if err := iso.Write(x.dir, *rec); err != nil {
			x.t.Errorf("write %s: %v", id, err)
		}
	}
	// All records are persisted before the aggregate test is failed.
	for _, id := range []string{"AT10", "AT11", "AT12", "AT13"} {
		if x.records[id].Status != iso.StatusPass {
			x.t.Errorf("%s: %s", id, x.records[id].Status)
		}
	}
}

func (x *isolationRun) add(id string, c iso.Check) {
	c.At = time.Now().UTC().Format(time.RFC3339)
	x.records[id].Checks = append(x.records[id].Checks, c)
	x.records[id].Status = iso.ChecksStatus(x.records[id].Checks)
	if err := iso.Write(x.dir, *x.records[id]); err != nil {
		x.t.Error(err)
	}
	x.t.Logf("%s %s: %s (positive=%t)", id, c.Name, c.Status, c.Positive.Valid())
}

func (x *isolationRun) proof(r *runner) iso.Positive {
	p := iso.Positive{DeviceID: r.usageDevice.String(), Container: r.lab.member, At: time.Now().UTC().Format(time.RFC3339)}
	// Wait only for advertisement; selection itself remains the shared driver.
	wait, cancel := context.WithTimeout(x.ctx, 45*time.Second)
	advertised := waitUntil(wait, 2*time.Second, func() bool {
		raw, _ := r.lab.docker(r.lab.member, "netbird", "networks", "list")
		return strings.Contains(string(raw), "fvpn-exit-")
	})
	cancel()
	if !advertised {
		p.Observations = append(p.Observations, iso.Observation{Name: "advertisement", Detail: "no product exit advertised within 45 seconds"})
		return p
	}
	p.Counted = r.countedFlow(x.ctx, func(k, v string) {
		p.Observations = append(p.Observations, iso.Observation{Name: k, Detail: v})
		if k == "admission_bytes_before" {
			p.Before = v
		}
		if k == "admission_bytes_after" {
			p.After = v
		}
	})
	if p.Valid() {
		x.noteFirstCounted(p)
	}
	return p
}

func (x *isolationRun) observation(c *iso.Check, r *runner) {
	heartbeat, _ := r.lab.docker(r.lab.node, "cat", "/run/vpn-node/heartbeat")
	c.Observations = append(c.Observations, iso.Observation{Name: "heartbeat", Detail: strings.TrimSpace(string(heartbeat))})
	conn, e := r.lab.db(x.ctx)
	if e != nil {
		return
	}
	defer conn.Close(x.ctx)
	var raw []byte
	if conn.QueryRow(x.ctx, `SELECT jsonb_build_object('generation',report_payload->'applied_config_generation','checks',report_payload->'checks','errors',report_payload->'errors','forwarding',report_payload->'member_forwarding_enabled','reasons',report_payload->'forwarding_block_reasons') FROM node_observations WHERE node_id=$1 ORDER BY received_at DESC LIMIT 1`, r.nodeID).Scan(&raw) == nil {
		c.Observations = append(c.Observations, iso.Observation{Name: "accepted_observation", Detail: string(raw)})
	}
}

func (x *isolationRun) prepare() bool {
	status, requestID, detail := x.putFleet(60, "TEST ONLY D-087 fixed offline window before enrollment")
	x.fleetRequestID = requestID
	x.extra("fleet-before-enrollment.json", map[string]any{"status": status, "request_id": requestID, "detail": detail, "offline_seconds": 60})
	if status != 200 {
		x.problem("w3-wire-nodes", "pre-enrollment fleet configuration failed")
		return false
	}

	baseline, err := PrepareServingBaseline(x.ctx, x.admin.http, x.admin.csrf, false)
	if err != nil {
		x.problem("w3-devstack-e2e", "PrepareServingBaseline: "+err.Error())
		return false
	}
	for _, rec := range x.records {
		rec.Observations = append(rec.Observations, iso.Observation{Name: "boot_release", Detail: baseline.Boot.String()})
	}
	r := x.r
	rules, _, err := r.lab.createPolicy(x.ctx, x.admin, "domain_rule_set", "isolation-allow", map[string]any{"rules": []any{map[string]any{"rule_id": uuid.NewString(), "domain": "feed-01.example.invalid", "include_subdomains": false, "action": "allow", "profile_id": nil, "reason": "admin_rule", "revision": "1"}}})
	if err != nil {
		x.problem("w3-wire-dns", err.Error())
		return false
	}
	r.profileID, _, err = r.lab.createPolicy(x.ctx, x.admin, "dns_profile", "isolation-profile", map[string]any{"profile_code": "isolation", "display_name": map[string]string{"zh-TW": "TEST ONLY"}, "block_response": "nxdomain", "blocked_ttl_seconds": 5, "domain_rule_set_revision_ids": []uuid.UUID{rules}, "threat_feed_revision_ids": []string{}})
	if err != nil {
		x.problem("w3-wire-dns", err.Error())
		return false
	}
	r.catalogue(x.ctx, x.admin)
	if r.planRevID == uuid.Nil {
		x.problem("w3-wire-members", r.planError)
		return false
	}
	created, e := x.admin.call(x.ctx, "POST", r.lab.public+"/api/v1/admin/groups", api.MemberGroupCreate{Code: "isolation-b", Name: "TEST ONLY isolation B"}, map[string]string{"Idempotency-Key": uuid.NewString()})
	var group api.MemberGroup
	if e != nil || created.Status != 201 || json.Unmarshal(created.Body, &group) != nil {
		x.problem("w3-wire-nodes", "group B creation failed")
		return false
	}
	x.groupB = group.GroupId
	conn, e := r.lab.db(x.ctx)
	if e != nil {
		return false
	}
	defer conn.Close(x.ctx)
	var raw []byte
	if conn.QueryRow(x.ctx, `SELECT payload FROM policy_revisions WHERE id=$1`, r.planRevID).Scan(&raw) != nil {
		return false
	}
	var payload map[string]json.RawMessage
	if json.Unmarshal(raw, &payload) != nil {
		return false
	}
	payload["exit_group_ids"], _ = json.Marshal([]uuid.UUID{x.groupB})
	x.planB, _, err = r.lab.createPolicy(x.ctx, x.admin, "plan", "isolation-plan-b", payload)
	if err != nil {
		x.problem("w3-wire-members", err.Error())
		return false
	}
	n, err := x.stack.Enroll(x.ctx, BrowserSession{x.admin.http, x.admin.csrf, x.admin.memberID}, "TEST ONLY isolation exit")
	if err != nil {
		x.problem("w2-agent-core", err.Error())
		return false
	}
	r.nodeID, r.incID = n.NodeID, n.IncarnationID
	x.startConvergence()
	r.publish(x.ctx, x.admin, n.NodeID)
	c := iso.Check{Name: "serving setup", Status: iso.StatusBlocked, Owner: "w2-agent-core / w3-wire-nodes"}
	if !x.publication("published", []uuid.UUID{r.groupID, x.groupB}, &c) {
		x.noteServingGap(&c)
		x.observation(&c, r)
		for _, id := range []string{"AT10", "AT11", "AT12", "AT13"} {
			x.add(id, c)
		}
		x.noteUnmeasured(c)
		return false
	}
	return true
}

// noteServingGap reads the node ruleset after a failed publication. It does
// not write nft. #220 fail-closes a foreign postrouting xt target as
// FLOWTABLE_BYPASS, and the agent reports only that code.
func (x *isolationRun) noteServingGap(c *iso.Check) {
	summary := "nft ruleset was not read"
	if x.r != nil && x.r.lab != nil {
		raw, err := x.r.lab.docker(x.r.lab.node, "nft", "-j", "list", "ruleset")
		if err != nil {
			summary = "nft -j list ruleset failed: " + clipText(strings.TrimSpace(string(raw)+" "+err.Error()), 400)
		} else {
			summary = foreignRulesetSummary(raw)
		}
	}
	c.Observations = append(c.Observations, iso.Observation{Name: "foreign_ruleset", Detail: summary}, iso.Observation{Name: "d079_classifier", Detail: d079Note()})
	switch {
	case dockerEmbeddedDNS(summary):
		c.Owner = "w3-devstack-e2e: node netns table ip nat POSTROUTING xt SNAT for 127.0.0.11"
		c.Actual = "serving publication fail-closed; Docker embedded DNS NAT is still in the node ruleset after follow-up 14; " + summary + "; the harness did not write nft"
	case strings.Contains(summary, "127.0.0.") || strings.Contains(summary, "loopback"):
		c.Owner = "w1-netlab: D-079 not merged; VerifyServing does not classify a loopback-scoped foreign rule"
		c.Actual = "serving publication fail-closed on a loopback-looking foreign rule; " + summary + "; the harness did not write nft"
	case strings.Contains(summary, "no foreign flowtable"):
		c.Actual = "publication did not complete; no foreign flowtable, xt statement, or base chain outside inet fvpn; " + summary
	default:
		c.Owner = "w3-devstack-e2e: new foreign table in the lab node"
		c.Actual = "serving publication fail-closed; foreign ruleset: " + summary + "; the harness did not write nft"
	}
}

func (x *isolationRun) noteRulesetReadback() {
	rec := x.records["AT10"]
	if rec == nil || x.r == nil || x.r.lab == nil {
		return
	}
	rec.Observations = append(rec.Observations, iso.Observation{Name: "d079_classifier", Detail: d079Note()})
	raw, err := x.r.lab.docker(x.r.lab.node, "nft", "-j", "list", "ruleset")
	if err != nil {
		rec.Observations = append(rec.Observations, iso.Observation{Name: "foreign_rules", Detail: "nft -j list ruleset failed: " + clipText(strings.TrimSpace(string(raw)+" "+err.Error()), 400)})
	} else {
		listing, dockerNAT := foreignRulesetListing(raw)
		class := "no foreign rule; Docker embedded DNS NAT absent; serving pass attributable to lab fix #240; D-079 #241 present but not exercised"
		if dockerNAT {
			class = "Docker embedded DNS NAT present; inspect the foreign rule scopes for D-079 attribution; observation report has no per-rule classification field"
		} else if listing != "[]" {
			class = "foreign rules listed verbatim; no Docker embedded DNS NAT; observation report has no per-rule classification field"
		}
		rec.Observations = append(rec.Observations,
			iso.Observation{Name: "foreign_rules", Detail: listing},
			iso.Observation{Name: "foreign_rule_classification", Detail: class},
		)
	}
	conn, e := x.r.lab.db(x.ctx)
	if e != nil {
		return
	}
	defer conn.Close(x.ctx)
	var payload []byte
	var observationID uuid.UUID
	if conn.QueryRow(x.ctx, `SELECT id, report_payload FROM node_observations WHERE node_id=$1 ORDER BY received_at DESC LIMIT 1`, x.r.nodeID).Scan(&observationID, &payload) == nil {
		rec.Observations = append(rec.Observations, iso.Observation{Name: "observation_report", Detail: string(payload)})
		var report map[string]json.RawMessage
		_ = json.Unmarshal(payload, &report)
		tc, ok := report["tc_evidence"]
		_, hasEvidence := report["tc_evidence"]
		meta, _ := json.MarshalIndent(map[string]any{
			"run_id": rec.RunID, "build": rec.Build, "observation_id": observationID.String(),
			"report_id": report["report_id"], "observed_at": report["observed_at"],
			"has_tc": ok, "has_serving_evidence": hasEvidence,
		}, "", "  ")
		if err := iso.WriteExtra(x.dir, "serving-readback-metadata.json", string(meta)+"\n"); err != nil {
			x.t.Error(err)
		}
		if ok {
			rec.Observations = append(rec.Observations, iso.Observation{Name: "tc_readback", Detail: string(tc)})
		} else {
			rec.Observations = append(rec.Observations, iso.Observation{Name: "tc_readback", Detail: "absent from product observation report; observation_id=" + observationID.String() + "; report_id=" + string(report["report_id"]) + "; owner w1-netlab / w2-agent-core; mechanism evidence is separately recorded in lab-AT10-tc.json"})
		}
	}
	// Retain the accepted report/issuance history as well as sampled state.
	// This distinguishes a missed short window from a bootstrap that never
	// emitted the provisional report. No node observation is synthesized.
	var history []byte
	if err := conn.QueryRow(x.ctx, `SELECT jsonb_build_object(
		'run_id', $2::text, 'build', $3::text,
		'captured_at', to_char(clock_timestamp() AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
		'scope', 'Read-only bootstrap history through first counted control completion; no synthetic observations submitted.',
		'awaiting_report_count', (SELECT count(*) FROM node_observations WHERE node_id=$1 AND report_payload->'forwarding_block_reasons' ? 'awaiting_member_grants'),
		'leases', (SELECT jsonb_agg(v ORDER BY issued_at) FROM (SELECT issued_at, jsonb_build_object(
			'lease_id', id, 'generation', policy_generation::text,
			'issued_at', to_char(issued_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
			'member_count', jsonb_array_length(envelope->'signed'->'members')) v FROM serving_leases WHERE node_id=$1) s),
		'reports', (SELECT jsonb_agg(v ORDER BY received_at) FROM (SELECT received_at, jsonb_build_object(
			'observation_id', id, 'report_id', report_id, 'observed_at', report_payload->'observed_at',
			'received_at', to_char(received_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
			'generation', report_payload->'applied_config_generation', 'provisioning', report_payload->'provisioning_state',
			'safety', report_payload->'safety_status', 'reasons', report_payload->'forwarding_block_reasons',
			'errors', report_payload->'errors') v FROM node_observations WHERE node_id=$1) s)
	)`, x.r.nodeID, rec.RunID, rec.Build).Scan(&history); err != nil {
		x.t.Error("bootstrap history read failed")
	} else if err := iso.WriteExtra(x.dir, "bootstrap-renewal-evidence.json", string(history)+"\n"); err != nil {
		x.t.Error(err)
	}
}

func moduleRoot() string {
	// The overlay test runs with working directory tests/e2e.
	for _, dir := range []string{".", "..", "../.."} {
		if _, err := os.Stat(filepath.Join(dir, "node/enforce/verify_foreign.go")); err == nil {
			abs, absErr := filepath.Abs(dir)
			if absErr == nil {
				return abs
			}
			return dir
		}
	}
	return "."
}

func d079Note() string {
	root := moduleRoot()
	out, err := exec.Command("git", "-C", root, "log", "-1", "--format=%H %s", "--", "node/enforce/verify_foreign.go").CombinedOutput()
	line := strings.TrimSpace(string(out))
	if err != nil {
		line = "git log failed: " + line
	}
	body, readErr := os.ReadFile(filepath.Join(root, "node/enforce/verify_foreign.go"))
	present := readErr == nil && strings.Contains(string(body), "loopback_scoped")
	return "verify_foreign.go " + line + "; loopback_scoped_in_source=" + boolText(present) + "; this main's observation report has no foreign-rule classification field"
}

func foreignRulesetListing(raw []byte) (string, bool) {
	var doc struct {
		Nftables []map[string]json.RawMessage `json:"nftables"`
	}
	if json.Unmarshal(raw, &doc) != nil {
		return "nft json unreadable", false
	}
	var kept []string
	dockerNAT := false
	for _, obj := range doc.Nftables {
		if _, ok := obj["metainfo"]; ok {
			continue
		}
		fam, table := nftObjectIdentity(obj)
		if table == "fvpn" && (fam == "inet" || fam == "") {
			continue
		}
		encoded, err := json.Marshal(obj)
		if err != nil {
			continue
		}
		text := string(encoded)
		if strings.Contains(text, "127.0.0.11") && strings.Contains(text, "DOCKER_") {
			dockerNAT = true
		}
		kept = append(kept, text)
	}
	if len(kept) == 0 {
		return "[]", dockerNAT
	}
	return "[" + strings.Join(kept, ",") + "]", dockerNAT
}

func nftObjectIdentity(obj map[string]json.RawMessage) (family, table string) {
	for key, raw := range obj {
		var id struct {
			Family string `json:"family"`
			Table  string `json:"table"`
			Name   string `json:"name"`
		}
		if json.Unmarshal(raw, &id) != nil {
			continue
		}
		// A table declaration stores its name in "name". Chains, rules and
		// counters store the owning table in "table".
		if key == "table" && id.Name != "" {
			return id.Family, id.Name
		}
		if id.Family != "" || id.Table != "" {
			return id.Family, id.Table
		}
	}
	return "", ""
}

func foreignRulesetSummary(raw []byte) string {
	var doc struct {
		Nftables []map[string]json.RawMessage `json:"nftables"`
	}
	if json.Unmarshal(raw, &doc) != nil {
		return "nft json unreadable"
	}
	var lines []string
	for _, obj := range doc.Nftables {
		if body, ok := obj["flowtable"]; ok {
			lines = append(lines, "flowtable "+clipText(string(body), 240))
		}
		if body, ok := obj["chain"]; ok {
			var c struct {
				Family string          `json:"family"`
				Table  string          `json:"table"`
				Name   string          `json:"name"`
				Hook   string          `json:"hook"`
				Policy string          `json:"policy"`
				Prio   json.RawMessage `json:"prio"`
			}
			if json.Unmarshal(body, &c) != nil || c.Hook == "" || (c.Family == "inet" && c.Table == "fvpn") {
				continue
			}
			lines = append(lines, fmt.Sprintf("chain %s %s %s hook %s prio %s policy %s", c.Family, c.Table, c.Name, c.Hook, strings.TrimSpace(string(c.Prio)), c.Policy))
		}
		if body, ok := obj["rule"]; ok {
			var rule struct {
				Family string                       `json:"family"`
				Table  string                       `json:"table"`
				Chain  string                       `json:"chain"`
				Expr   []map[string]json.RawMessage `json:"expr"`
			}
			if json.Unmarshal(body, &rule) != nil || (rule.Family == "inet" && rule.Table == "fvpn") {
				continue
			}
			for _, expr := range rule.Expr {
				for _, key := range []string{"xt", "flow", "snat", "dnat", "masquerade", "mangle", "jump"} {
					stmt, ok := expr[key]
					if !ok {
						continue
					}
					lines = append(lines, fmt.Sprintf("rule %s %s %s %s %s", rule.Family, rule.Table, rule.Chain, key, clipText(string(stmt), 160)))
				}
				if match, ok := expr["match"]; ok && strings.Contains(string(match), "127.0.0.11") {
					lines = append(lines, fmt.Sprintf("rule %s %s %s match %s", rule.Family, rule.Table, rule.Chain, clipText(string(match), 160)))
				}
			}
		}
		if len(lines) >= 40 {
			break
		}
	}
	if len(lines) == 0 {
		return "no foreign flowtable, xt statement, or base chain outside inet fvpn"
	}
	return clipText(strings.Join(lines, "; "), 4000)
}

func (x *isolationRun) publication(intent string, groups []uuid.UUID, c *iso.Check) bool {
	note := func(k, v string) {
		if c != nil {
			c.Observations = append(c.Observations, iso.Observation{Name: k, Detail: v})
		}
	}
	// Long capture matrices can outlive a browser session. Obtain a fresh one
	// before each mutation; never replay an unasserted side effect.
	var loginErr error
	x.admin, loginErr = x.r.lab.login(x.ctx, "e2e-admin", "admin")
	if loginErr != nil {
		note("publication_login", "failed")
		return false
	}
	path := x.r.lab.public + "/api/v1/admin/nodes/" + x.r.nodeID.String()
	get, e := x.admin.call(x.ctx, "GET", path, nil, nil)
	if e != nil || get.Status != 200 {
		return false
	}
	put, e := x.admin.call(x.ctx, "PUT", path+"/publication", map[string]any{"publication_intent": intent, "group_ids": groups, "reason": "TEST ONLY isolation publication"}, map[string]string{"Idempotency-Key": uuid.NewString(), "If-Match": get.Header.Get("ETag")})
	note("publication_put", httpDetail(put, e))
	if e != nil || put.Status != 202 {
		note("publication_error", apiErrorText(put.Body))
		return false
	}
	id := operationID(put.Body)
	wait, cancel := context.WithTimeout(x.ctx, 150*time.Second)
	defer cancel()
	var last reply
	status := ""
	done := waitUntil(wait, 2*time.Second, func() bool {
		last, e = x.admin.call(wait, "GET", x.r.lab.public+"/api/v1/admin/operations/"+id, nil, nil)
		var v opView
		if e != nil || last.Status != 200 || json.Unmarshal(last.Body, &v) != nil {
			return false
		}
		if status != v.Status {
			note("operation_transition", time.Now().UTC().Format(time.RFC3339Nano)+" "+v.Status)
		}
		status = v.Status
		// Central withdrawal may be confirmed before the agent ACK. Its
		// partially_succeeded operation still accepts observation-driven
		// completion; keep the disabled intent until that read-back arrives.
		return status == "succeeded" || status == "failed"
	})
	note("operation", string(last.Body))
	node, _ := x.admin.call(x.ctx, "GET", path, nil, nil)
	var detail map[string]json.RawMessage
	_ = json.Unmarshal(node.Body, &detail)
	for _, key := range []string{"publication_intent", "publication_observed", "provisioning_state", "safety_status", "checks", "config_generation"} {
		note(key, string(detail[key]))
	}
	return done && status == "succeeded"
}

func (x *isolationRun) join(subject string, plan uuid.UUID, name string, until time.Time) *runner {
	member, e := x.stack.Login(x.ctx, subject, "member")
	if e != nil {
		return nil
	}
	resp, e := x.stack.Call(x.ctx, BrowserSession{x.admin.http, x.admin.csrf, x.admin.memberID}, "POST", "/api/v1/admin/members/"+member.MemberID+"/grants", api.GrantCreateRequest{Source: api.GrantSourceAdminGrant, Reason: "TEST ONLY finite isolation grant", PlanRevisionId: &plan, ValidUntil: until}, map[string]string{"Idempotency-Key": uuid.NewString()})
	if e != nil || resp.Status != 202 {
		x.t.Log("grant", httpDetail(resp, e))
		return nil
	}
	container, e := x.stack.StartNativePeer(x.ctx, name)
	if e != nil {
		x.t.Log(e)
		return nil
	}
	peer, e := x.stack.JoinNativeMember(x.ctx, member, subject, container, container)
	if e != nil {
		x.t.Log(e)
		return nil
	}
	r := *x.r
	l := *x.r.lab
	r.lab = &l
	r.lab.member = container
	r.usageDevice = peer.DeviceID
	return &r
}

func (x *isolationRun) connect(r *runner, ip string, port int) (string, string) {
	route, _ := r.lab.docker(r.lab.member, "ip", "route", "get", ip)
	raw, e := r.lab.docker(r.lab.member, "/tmp/isolation-probe", "connect", ip, strconv.Itoa(port))

		Result string `json:"result"`
	}

	}

}

// Capture only fixture headers at both boundaries. A route and a timeout alone
// cannot attribute a denial to the exit's enforcement.
func (x *isolationRun) capturedConnect(r *runner, target, ip string, port int, c *iso.Check) (route, result string, ingress, delivered bool) {
	type capture struct {
		name    string
		cmd     *exec.Cmd
		out     bytes.Buffer
		started bool
	}
	captures := []*capture{}
	for _, v := range []struct{ name, container, iface string }{{"negative_node_ingress", r.lab.node, "fvmb0"}, {"negative_target_ingress", target, "eth0"}} {
		direction := "in"
		if v.container == r.lab.prefix+"-origin" {
			// The service image has no tcpdump. Observe the origin bridge at
			// the node's uplink instead, using the shared node image's tools.
			v.name, v.container, v.iface, direction = "negative_node_egress", r.lab.node, "fvup0", "out"
		}
		args := []string{"exec", v.container, "timeout", "-s", "INT", "3", "tcpdump", "-nn", "-q", "-s", "96", "-Q", direction, "-i", v.iface, "dst", "host", ip, "and", "tcp", "dst", "port", strconv.Itoa(port)}
		if v.name == "negative_node_ingress" {
			args = append(args, "and", "src", "host", memberIP(r))
		}
		cap := &capture{name: v.name, cmd: exec.CommandContext(x.ctx, "docker", args...)}
		cap.cmd.Stdout, cap.cmd.Stderr = &cap.out, &cap.out
		cap.started = cap.cmd.Start() == nil
		captures = append(captures, cap)
	}
	time.Sleep(time.Second)
	route, result = x.connect(r, ip, port)
	valid := true
	for _, cap := range captures {
		if cap.started {
			_ = cap.cmd.Wait()
		}
		raw := cap.out.String()
		c.Observations = append(c.Observations, iso.Observation{Name: cap.name, Detail: strings.TrimSpace(raw)})
		n, parsed := iso.CaptureCount(raw)
		ok := cap.started && parsed
		valid = valid && ok
		if ok {
			if cap.name == "negative_node_ingress" {
				ingress = n > 0
			} else {
				delivered = n > 0
			}
		}
	}
	return route, result, ingress && valid, delivered
}

	}
	_ = json.Unmarshal(x.r.desiredRaw(), &desired)
	if len(desired.Signed.DNS.Resolvers) > 0 {
		resolver := desired.Signed.DNS.Resolvers[0]
		route, _ := x.a1.lab.docker(x.a1.lab.member, "ip", "route", "get", resolver)

			if err != nil {
				c.Observations = append(c.Observations, iso.Observation{Name: "error", Detail: err.Error()})
			}
		}
		if c.Positive.Valid() {
			c.Status = iso.StatusFail
			if ok {
				c.Status = iso.StatusPass
				c.Owner = ""
			}
		}
		x.add("AT12", c)
	}
	x.spoofedSource()
}

}

}

func (x *isolationRun) restart() {
	c := iso.Check{Name: "node restart with durable identity", Positive: x.proof(x.a1), Owner: "w2-agent-core / w3-devstack-e2e"}
	before, _ := x.r.lab.docker(x.r.lab.node, "cat", "/run/vpn-node/heartbeat")
	boot := regexp.MustCompile(`(?m)^boot_id (\S+)$`).FindStringSubmatch(string(before))
	x.retainReasonLogs("before-restart")
	out, e := exec.Command("docker", "restart", x.r.lab.node).CombinedOutput()
	_ = exec.Command("docker", "exec", "-d", x.r.lab.node, "/tmp/isolation-probe", "nft-clock").Run()
	wait, cancel := context.WithTimeout(x.ctx, 90*time.Second)
	ready := waitUntil(wait, 2*time.Second, func() bool {
		raw, _ := x.r.lab.docker(x.r.lab.node, "cat", "/run/vpn-node/heartbeat")
		// A container process restart retains the kernel boot ID. Require a
		// fresh heartbeat as well as serving read-back; this is not OS reboot evidence.
		return len(boot) == 2 && strings.Contains(string(raw), "boot_id "+boot[1]) && !bytes.Equal(raw, before) && strings.Contains(string(raw), "\nserving 1\n") && x.r.lab.forwardChain().MemberAccept > 0
	})
	cancel()
	after := x.proof(x.a1)
	for attempt, deadline := 0, time.Now().Add(60*time.Second); !after.Valid() && time.Now().Before(deadline); attempt++ {
		raw, _ := json.Marshal(after)
		c.Observations = append(c.Observations, iso.Observation{Name: "recovery_attempt_" + strconv.Itoa(attempt), Detail: string(raw)})
		time.Sleep(3 * time.Second)
		after = x.proof(x.a1)
	}
	afterRaw, _ := json.Marshal(after)
	c.Observations = append(c.Observations, iso.Observation{Name: "recovered_control", Detail: string(afterRaw)})
	c.Status = iso.NegativeStatus(c.Positive, e == nil && ready && after.Valid(), true)
	c.Actual = "container restart=" + strings.TrimSpace(string(out)) + "; recovered counted=" + boolText(after.Valid()) + "; continuous forbidden-flow coverage not established"
	clockRaw, clockErr := x.r.lab.docker(x.r.lab.node, "/tmp/isolation-probe", "producer-clock")
	if clockErr != nil || !json.Valid(clockRaw) {
		x.t.Error("post-restart producer clock unavailable")
	} else {
		x.extra("producer-clock-after-restart.json", json.RawMessage(clockRaw))
	}
	x.observation(&c, x.a1)
	x.add("AT13", c)
	x.add("AT13", iso.Check{Name: "continuous crash/NetBird/reboot/updater transitions", Status: iso.StatusBlocked, Positive: after, Actual: "restart recovery measured; full continuous fault matrix remains unproved", Owner: "acceptance / operator"})
}

func (x *isolationRun) deadline() {
	defer x.wholeNodeExpiry()
	x.admin, _ = x.r.lab.login(x.ctx, "e2e-admin", "admin")
	if x.admin == nil {
		return
	}
	end := time.Now().UTC().Add(180 * time.Second).Truncate(time.Second)
	r := x.join("iso-expiry", x.r.planRevID, "expiry", end)
	if r == nil {
		x.add("AT13", iso.Check{Name: "lease member expiry", Status: iso.StatusBlocked, Actual: "finite member admission failed", Owner: "w3-wire-members"})
		return
	}
	if !x.install(r.lab.member) {
		return
	}
	c := iso.Check{Name: "lease member expiry under control loss", Positive: x.proof(r), Owner: "w2-agent-core"}
	conn, e := r.lab.db(x.ctx)
	if e != nil {
		return
	}
	defer conn.Close(x.ctx)
	var lease []byte
	binding, bound := readProjectedBinding(r.desiredRaw(), r.usageDevice)
	var bounds struct {
		Expires time.Time `json:"expires_at"`
		Members []struct {
			Member     uuid.UUID `json:"member_id"`
			End        time.Time `json:"lease_end"`
			ServiceEnd time.Time `json:"effective_service_end"`
		} `json:"members"`
	}
	// With a fixed 60-second node window, an early lease correctly expires
	// before this finite grant. Wait for ordinary renewal to reach the member
	// deadline while the node lease still extends beyond it. Never edit either.
	wait, cancel := context.WithDeadline(x.ctx, end.Add(-5*time.Second))
	samples := []map[string]any{}
	leaseBound := waitUntil(wait, time.Second, func() bool {
		e = conn.QueryRow(wait, `SELECT jsonb_build_object('lease_id',id,'issued_at',issued_at,'expires_at',expires_at,'generation',policy_generation::text,'members',envelope->'signed'->'members',
            'accepted_serving',EXISTS(SELECT 1 FROM node_observations o WHERE o.node_id=s.node_id AND o.report_payload->'lease'->>'lease_id'=s.id::text AND o.report_payload->>'member_forwarding_enabled'='true'))
            FROM serving_leases s WHERE node_id=$1 ORDER BY issued_at DESC LIMIT 1`, r.nodeID).Scan(&lease)
		if e != nil || json.Unmarshal(lease, &bounds) != nil {
			return false
		}
		samples = append(samples, map[string]any{"at": time.Now().UTC(), "lease": json.RawMessage(append([]byte(nil), lease...))})
		var receipt struct {
			Accepted bool `json:"accepted_serving"`
		}
		if json.Unmarshal(lease, &receipt) != nil || !receipt.Accepted || !bound || !bounds.Expires.After(end.Add(25*time.Second)) {
			return false
		}
		for _, m := range bounds.Members {
			if m.Member == binding.Member && m.ServiceEnd.Equal(end) && m.End.Equal(end) {
				return true
			}
		}
		return false
	})
	cancel()
	x.extra("finite-member-renewal-wait.json", map[string]any{"run_id": x.records["AT13"].RunID, "member_deadline": end, "samples": samples, "qualified": leaseBound})
	c.Observations = append(c.Observations, iso.Observation{Name: "signed_lease_bounds", Detail: string(lease)}, iso.Observation{Name: "grant_deadline", Detail: end.Format(time.RFC3339)})
	if !c.Positive.Valid() || e != nil || !leaseBound || !time.Now().Before(end.Add(-5*time.Second)) {
		c.Status = iso.StatusBlocked
		c.Actual = "counted finite flow, matching signed member deadline, or pre-expiry fault window unavailable"
		x.add("AT13", c)
		return
	}
	for time.Until(end) > 5*time.Second {
		select {
		case <-x.ctx.Done():
			return
		case <-time.After(time.Second):
		}
	}
	e = exec.Command("docker", "stop", "--time", "2", r.lab.control, r.lab.prefix+"-worker").Run()
	c.Observations = append(c.Observations, iso.Observation{Name: "control_stopped_at", Detail: time.Now().UTC().Format(time.RFC3339)})
	defer func() { _ = exec.Command("docker", "start", r.lab.control, r.lab.prefix+"-worker").Run() }()
	if e != nil || !time.Now().Before(end) {
		c.Status = iso.StatusBlocked
		c.Actual = "control stop did not complete before the finite member deadline"
		x.add("AT13", c)
		return
	}
	for time.Now().Before(end.Add(2 * time.Second)) {
		select {
		case <-x.ctx.Done():
			return
		case <-time.After(time.Second):
		}
	}
	frozen, available := r.admissionBytes()
	denied := true
	routed, probesOK := true, true
	for i := 0; i < 5; i++ {
		route, result, ingress, delivered := x.capturedConnect(r, r.lab.prefix+"-origin", r.originIP(), 443, &c)
		c.Observations = append(c.Observations, iso.Observation{Name: "post_deadline_" + strconv.Itoa(i), Detail: "result=" + result + "; route=" + route})
		denied = denied && result == "timeout" && !delivered
		routed = routed && strings.Contains(route, "dev fvnb0")
		probesOK = probesOK && result != "probe_error" && ingress
		time.Sleep(time.Second)
	}
	after, ok := r.admissionBytes()
	c.Observations = append(c.Observations, iso.Observation{Name: "post_deadline_bytes_before", Detail: fmt.Sprint(frozen)}, iso.Observation{Name: "post_deadline_bytes_after", Detail: fmt.Sprint(after)})
	// All five member-denial probes must finish while the original node lease
	// is still valid, so whole-node expiry cannot supply this result.
	nodeStillValid := time.Now().Before(bounds.Expires)
	c.Observations = append(c.Observations, iso.Observation{Name: "node_lease_valid_through_last_probe", Detail: boolText(nodeStillValid) + "; completed=" + time.Now().UTC().Format(time.RFC3339Nano) + "; node_expires=" + bounds.Expires.UTC().Format(time.RFC3339Nano)})
	c.Status = iso.NegativeStatus(c.Positive, available && ok && probesOK && nodeStillValid, denied && after == frozen)
	c.Actual = "five new TCP requests after finite signed member deadline; control and worker stopped; admission bytes frozen=" + boolText(after == frozen) + "; route retained throughout=" + boolText(routed) + "; node lease has a later deadline"
	x.observation(&c, r)
	x.add("AT13", c)
	if !routed {
		x.add("AT13", iso.Check{Name: "expiry per-packet ingress attribution", Status: iso.StatusBlocked, Positive: c.Positive, Actual: "Native route was withdrawn during the post-deadline window. End-to-end withdrawal was measured; persistent post-expiry ingress was not.", Owner: "w3-devstack-e2e: retain controlled ingress after authorization withdrawal"})
	}
}

func (x *isolationRun) wholeNodeExpiry() {
	if x == nil || x.r == nil || x.a1 == nil {
		return
	}
	c := iso.Check{Name: "short whole-node lease expiry", Status: iso.StatusBlocked, Owner: "w2-agent-core", Actual: "not measured"}
	var stopped bool
	defer func() {
		if stopped {
			_ = exec.Command("docker", "start", x.r.lab.control, x.r.lab.prefix+"-worker").Run()
		}
		if stopped && !x.waitControl() {
			c.Status = iso.StatusBlocked
			c.Actual += "; control restart incomplete"
		}
		if !x.fleetAudits(&c, x.fleetRequestID) {
			c.Status = iso.StatusBlocked
			c.Actual += "; pre-enrollment fleet audit missing"
		}
		x.add("AT13", c)
	}()
	admin, err := x.r.lab.login(x.ctx, "e2e-admin", "admin")
	if err != nil {
		c.Actual = "admin login failed before fleet-config write"
		c.Owner = "w3-devstack-e2e"
		return
	}
	x.admin = admin
	route, _ := x.a1.lab.docker(x.a1.lab.member, "ip", "route", "get", x.a1.originIP())
	if !strings.Contains(string(route), "dev fvnb0") {
		selected := x.a1.selectExit(x.ctx, func(k, v string) {
			c.Observations = append(c.Observations, iso.Observation{Name: "pre_lease_" + k, Detail: clipText(v, 400)})
		})
		route, _ = x.a1.lab.docker(x.a1.lab.member, "ip", "route", "get", x.a1.originIP())
		if !selected || !strings.Contains(string(route), "dev fvnb0") {
			c.Actual = "member route via fvnb0 unavailable before the short lease"
			c.Owner = "w3-devstack-e2e"
			return
		}
	}
	c.Observations = append(c.Observations, iso.Observation{Name: "fixed_offline_window", Detail: "60 seconds configured before enrollment; unchanged for this entire run"})
	putAt := time.Now().UTC().Add(-2 * time.Second)
	nodePath := x.r.lab.public + "/api/v1/admin/nodes/" + x.r.nodeID.String()
	get, getErr := x.admin.call(x.ctx, "GET", nodePath, nil, nil)
	patch, patchErr := x.admin.call(x.ctx, "PATCH", nodePath, map[string]any{"capacity": map[string]any{"down_bps": 204000000, "up_bps": 200000000, "member_down_bps_cap": 100000000, "member_up_bps_cap": 100000000}}, map[string]string{"If-Match": get.Header.Get("ETag"), "Idempotency-Key": uuid.NewString()})
	c.Observations = append(c.Observations, iso.Observation{Name: "renewal_capacity_patch", Detail: httpDetail(patch, patchErr) + " get=" + httpDetail(get, getErr)})
	wait, cancel := context.WithTimeout(x.ctx, 90*time.Second)
	var held leaseSnap
	found := waitUntil(wait, time.Second, func() bool {
		snap, ok := x.centralLeaseAfter(putAt)
		if !ok || snap.Max != 60 || snap.Members < 1 {
			return false
		}
		chain := x.r.lab.forwardChain()
		gen, _ := strconv.Atoi(snap.Generation)
		if chain.MemberAccept > 0 && !chain.Baseline && chain.Generation == gen && gen > 0 {
			held = snap
			return true
		}
		return false
	})
	cancel()
	if !found {
		c.Actual = "no applied 60s lease with member grants after the capacity renewal"
		c.Owner = "w2-agent-core / w3-wire-nodes"
		return
	}
	// Retain a continuous two-renewal window before the control-loss deadline.
	renewalCap := x.continuousCapture(x.r.lab.node, "any", "host "+x.a1.originIP()+" and udp port 8080", "renewal-node-capture.txt", 75)
	_ = exec.Command("docker", "exec", "-d", x.r.lab.node, "/tmp/isolation-probe", "packet-clock", "74", "/tmp/isolation-renewal-packets.jsonl").Run()
	_ = exec.Command("docker", "exec", "-d", x.a1.lab.member, "/tmp/isolation-probe", "renewal-probes", x.a1.originIP(), "1", "/tmp/isolation-renewal-probes.jsonl").Run()
	x.sleep(71 * time.Second)
	x.finishCapture(renewalCap)
	for _, item := range []struct{ container, path, name string }{{x.r.lab.node, "/tmp/isolation-renewal-packets.jsonl", "renewal-packets.json"}, {x.a1.lab.member, "/tmp/isolation-renewal-probes.jsonl", "renewal-probes.json"}} {
		raw, _ := exec.Command("docker", "exec", item.container, "cat", item.path).Output()
		rows := []json.RawMessage{}
		for _, line := range bytes.Split(raw, []byte("\n")) {
			if json.Valid(line) {
				rows = append(rows, append(json.RawMessage(nil), line...))
			}
		}
		x.extra(item.name, rows)
	}
	if err = exec.Command("docker", "stop", "--time", "2", x.r.lab.control, x.r.lab.prefix+"-worker").Run(); err != nil {
		c.Actual = "control and worker stop failed"
		c.Owner = "acceptance"
		return
	}
	stopped = true
	c.Observations = append(c.Observations, iso.Observation{Name: "control_stopped_at", Detail: time.Now().UTC().Format(time.RFC3339)})
	if nodeHeld, ok := x.readNodeLease(); ok {
		nodeHeld.Source = "node-store"
		held = nodeHeld
		c.Observations = append(c.Observations, iso.Observation{Name: "node_lease", Detail: mustJSON(held)})
	} else {
		held.Source = "central"
		c.Observations = append(c.Observations, iso.Observation{Name: "node_lease", Detail: "node store unread; using the applied central lease " + mustJSON(held)})
	}
	if held.Max != 60 || held.Members < 1 || held.Issued.Before(putAt) {
		c.Actual = "node does not hold a post-change 60s lease with member grants: " + mustJSON(held)
		c.Owner = "w2-agent-core / w3-wire-nodes"
		return
	}
	if time.Until(held.Expires) < 12*time.Second {
		c.Actual = "signed 60s deadline is inside 12s; measurement window missed: " + mustJSON(held)
		c.Owner = "w2-agent-core"
		return
	}
	earlyOK, before, after, earlyDetail := x.countedForward(x.a1)
	c.Observations = append(c.Observations, iso.Observation{Name: "forward_after_control_loss", Detail: earlyDetail})
	c.Positive = iso.Positive{DeviceID: x.a1.usageDevice.String(), Container: x.a1.lab.member, At: time.Now().UTC().Format(time.RFC3339), Counted: earlyOK, Before: strconv.FormatUint(before, 10), After: strconv.FormatUint(after, 10)}
	if !c.Positive.Valid() {
		c.Actual = "counted forward under the held 60s lease did not increase admission bytes; " + earlyDetail
		c.Owner = "w2-agent-core"
		return
	}
	if !x.sleep(time.Until(held.Expires.Add(-10 * time.Second))) {
		return
	}
	stopCapture := x.startContinuousExpiry(held)
	if !x.sleep(time.Until(held.Expires.Add(-4 * time.Second))) {
		return
	}
	nearOK, _, _, nearDetail := x.countedForward(x.a1)
	c.Observations = append(c.Observations, iso.Observation{Name: "forward_near_deadline", Detail: nearDetail}, iso.Observation{Name: "signed_deadline", Detail: held.Expires.Format(time.RFC3339Nano)})
	tick := x.agentTick()
	c.Observations = append(c.Observations, iso.Observation{Name: "agent_tick", Detail: tick.String()})
	var earlyBaseline, saw time.Time
	var reads []map[string]any
	for {
		now := time.Now()
		chain := x.r.lab.forwardChain()
		readEnd := time.Now()
		beat, _ := x.r.lab.docker(x.r.lab.node, "cat", "/run/vpn-node/heartbeat")
		reads = append(reads, map[string]any{"read_start": now.UTC().Format(time.RFC3339Nano), "read_end": readEnd.UTC().Format(time.RFC3339Nano), "heartbeat_read_end": time.Now().UTC().Format(time.RFC3339Nano), "chain": chain, "heartbeat": strings.TrimSpace(string(beat))})
		baselined := chain.Present && chain.Baseline && chain.Generation == 0 && chain.MemberAccept == 0 && chain.Policy == "drop"
		if baselined && now.Before(held.Expires) && earlyBaseline.IsZero() {
			earlyBaseline = now
		}
		if baselined && !now.Before(held.Expires) && saw.IsZero() {
			saw = now
			break
		}
		if !now.Before(held.Expires.Add(tick)) {
			break
		}
		if !x.sleep(200 * time.Millisecond) {
			return
		}
	}
	heartbeat, _ := x.r.lab.docker(x.r.lab.node, "cat", "/run/vpn-node/heartbeat")
	c.Observations = append(c.Observations, iso.Observation{Name: "baseline_at", Detail: saw.UTC().Format(time.RFC3339Nano)}, iso.Observation{Name: "early_baseline_at", Detail: earlyBaseline.UTC().Format(time.RFC3339Nano)}, iso.Observation{Name: "heartbeat_after_deadline", Detail: strings.TrimSpace(string(heartbeat))})
	x.extra("whole-node-bracketed-reads.json", map[string]any{"signed_deadline": held.Expires, "reads": reads})
	frozen, ok1 := x.a1.admissionBytes()
	dropRoute, dropResult, ingress, delivered := x.capturedConnect(x.a1, x.a1.lab.prefix+"-origin", x.a1.originIP(), 443, &c)
	frozenAfter, ok2 := x.a1.admissionBytes()
	flat := ok1 && ok2 && frozenAfter == frozen
	verdict := stopCapture()
	x.extra("deadline-verdict.json", verdict)
	onTime := verdict.Passed
	drop := dropResult == "timeout" && ingress && !delivered && strings.Contains(dropRoute, "dev fvnb0") && flat
	c.Observations = append(c.Observations, iso.Observation{Name: "post_deadline_forward", Detail: "result=" + dropResult + "; route=" + dropRoute + "; flat=" + boolText(flat) + "; bytes=" + fmt.Sprint(frozen) + "->" + fmt.Sprint(frozenAfter)})
	c.Actual = "60s lease held after control loss; near-deadline forward=" + boolText(nearOK) + "; D-104 exact packet deadline verdict=" + boolText(onTime) + "; last offset ns=" + verdict.LastPositiveOffset + "; first drop offset ns=" + verdict.FirstDropOffset + "; ingress without uplink=" + boolText(drop)
	if !nearOK || !onTime || !drop {
		c.Status = iso.StatusFail
		c.Owner = "w2-agent-core"
		return
	}
	c.Status = iso.StatusPass
	c.Owner = ""
}

func (x *isolationRun) putFleet(seconds int, reason string) (int, string, string) {
	if x.admin == nil {
		return 0, "", "no admin session"
	}
	path := x.r.lab.public + "/api/v1/admin/fleet-config"
	get, err := x.admin.call(x.ctx, "GET", path, nil, nil)
	if err != nil || get.Status != 200 {
		return get.Status, "", httpDetail(get, err) + " " + apiErrorText(get.Body)
	}

func (x *isolationRun) agentTick() time.Duration {
	raw, err := x.r.lab.docker(x.r.lab.node, "cat", "/etc/vpn-node/agent-run.json")
	var cfg struct {
		Heartbeat int `json:"heartbeat_seconds"`
		Poll      int `json:"poll_seconds"`
	}
	if err != nil || json.Unmarshal(raw, &cfg) != nil {
		return 5 * time.Second
	}
	tick := cfg.Heartbeat
	if cfg.Poll > 0 && (tick <= 0 || cfg.Poll < tick) {

	}
	if tick <= 0 {
		tick = 5
	}
	return time.Duration(tick) * time.Second
}

		return false
	case <-timer.C:
		return true
	}
}

func (x *isolationRun) spoofedSource() {
	positive := x.proof(x.a1)
	c := iso.Check{Name: "unknown/spoofed member source", Positive: positive, Owner: "w2-agent-core"}
	live := memberIP(x.a1)
	forged := forgedMemberSource
	if forged == live || forged == memberIP(x.a2) || forged == memberIP(x.b1) {
		c.Status = iso.StatusBlocked
		c.Owner = "w3-devstack-e2e"
		c.Actual = "forged source collided with a live overlay address"
		x.add("AT12", c)
		return
	}
	dest := x.a1.originIP()
	denied, derr := x.stack.ObserveMemberIngress(x.ctx, forged, dest)
	admitted, aerr := x.stack.ObserveMemberIngress(x.ctx, live, dest)
	up := "u_" + strings.ReplaceAll(x.a1.usageDevice.String(), "-", "")
	down := "d_" + strings.ReplaceAll(x.a1.usageDevice.String(), "-", "")
	c.Observations = append(c.Observations,
		iso.Observation{Name: "forged_inject", Detail: ingressNote(denied, derr)},
		iso.Observation{Name: "admitted_inject", Detail: ingressNote(admitted, aerr)},
	)
	if derr != nil || aerr != nil {
		c.Status = iso.StatusBlocked
		c.Owner = "w3-devstack-e2e"
		c.Actual = "ObserveMemberIngress failed; forged_err=" + errText(derr) + "; admitted_err=" + errText(aerr)
		x.add("AT12", c)
		return
	}
	deniedOK := ingressDenied(denied)
	admittedOK := ingressAdmitted(admitted, up, down)
	c.Status = iso.NegativeStatus(positive, true, deniedOK && admittedOK)
	if c.Status == iso.StatusPass {
		c.Owner = ""
	}
	c.Actual = "forged " + forged + " ctunku+1 and zero forwarded=" + boolText(deniedOK) + "; admitted " + up + " +1=" + boolText(admittedOK)
	x.add("AT12", c)
}

func (x *isolationRun) postRevokeIngress(revokedSource string, revokedDevice uuid.UUID, sibling *runner) {
	positive := x.proof(sibling)
	c := iso.Check{Name: "post-revoke kernel ingress attribution", Positive: positive, Owner: "w2-agent-core"}
	dest := sibling.originIP()
	revoked, rerr := x.stack.ObserveMemberIngress(x.ctx, revokedSource, dest)
	siblingSource := memberIP(sibling)
	admitted, aerr := x.stack.ObserveMemberIngress(x.ctx, siblingSource, dest)
	revokedUp := "u_" + strings.ReplaceAll(revokedDevice.String(), "-", "")
	up := "u_" + strings.ReplaceAll(sibling.usageDevice.String(), "-", "")
	down := "d_" + strings.ReplaceAll(sibling.usageDevice.String(), "-", "")
	c.Observations = append(c.Observations,
		iso.Observation{Name: "revoked_inject", Detail: ingressNote(revoked, rerr)},
		iso.Observation{Name: "admitted_inject", Detail: ingressNote(admitted, aerr)},
	)
	if rerr != nil || aerr != nil {
		c.Status = iso.StatusBlocked
		c.Owner = "w3-devstack-e2e"
		c.Actual = "ObserveMemberIngress failed after revoke; revoked_err=" + errText(rerr) + "; admitted_err=" + errText(aerr)
		x.add("AT13", c)
		return
	}
	revokedOK := ingressDenied(revoked) && revoked.Before[revokedUp] == revoked.After[revokedUp]
	admittedOK := ingressAdmitted(admitted, up, down)
	c.Status = iso.NegativeStatus(positive, revokedSource != "" && siblingSource != "", revokedOK && admittedOK)
	if c.Status == iso.StatusPass {
		c.Owner = ""
	}
	c.Actual = "revoked " + revokedSource + " ctunku+1 and zero forwarded=" + boolText(revokedOK) + "; sibling " + up + " +1=" + boolText(admittedOK)
	x.add("AT13", c)
}

func ingressHook(p MemberIngressProof) bool {
	return p.Mechanism == "tun-mirred" && p.LinkType == 65534 && p.MemberIface == "fvmb0" && p.SourcePort == 40000 && p.DestPort == 9
}

func ingressDenied(p MemberIngressProof) bool {
	if !ingressHook(p) || !captureEmpty(p.UplinkCapture) || !captureEmpty(p.OriginCapture) {
		return false
	}
	ct, ok := ingressCounterDelta(p.Before, p.After, "ctunku")
	return ok && ct.Packets == 1 && ct.Bytes > 0 && deviceCountersHeld(p.Before, p.After)
}

func ingressAdmitted(p MemberIngressProof, up, down string) bool {
	if !ingressHook(p) || up == "" || !captureSawDatagram(p.UplinkCapture) || !captureSawDatagram(p.OriginCapture) {
		return false
	}
	got, ok := ingressCounterDelta(p.Before, p.After, up)
	ct, ctOK := ingressCounterDelta(p.Before, p.After, "ctunku")
	return ok && got.Packets == 1 && got.Bytes > 0 && ctOK && ct.Packets == 0 && deviceCountersHeld(p.Before, p.After, up, down)
}

func ingressNote(p MemberIngressProof, err error) string {
	ct, _ := ingressCounterDelta(p.Before, p.After, "ctunku")
	raw, _ := json.Marshal(map[string]any{
		"error": errText(err), "mechanism": p.Mechanism, "link_type": p.LinkType, "member_iface": p.MemberIface,
		"source": p.Source, "dest": p.Dest, "source_port": p.SourcePort, "dest_port": p.DestPort,
		"ctunku_packets": ct.Packets, "ctunku_bytes": ct.Bytes, "deltas": p.Deltas,
		"uplink_capture": clipText(p.UplinkCapture, 800), "origin_capture": clipText(p.OriginCapture, 800),
	})
	return string(raw)
}

func errText(err error) string {
	if err == nil {
		return ""
	}
	return clipText(err.Error(), 300)
}

func (x *isolationRun) startConvergence() {
	if x.stopConv != nil || x.r == nil {
		return
	}
	x.stopConv = make(chan struct{})
	x.convStopped = make(chan struct{})
	go x.sampleLoop()
}

func (x *isolationRun) signalConv() {
	x.convOnce.Do(func() {
		if x.stopConv != nil {
			close(x.stopConv)
		}
	})
}

func (x *isolationRun) stopConvergence() {
	if x.stopConv == nil {
		return
	}
	x.signalConv()
	select {
	case <-x.convStopped:
	case <-time.After(15 * time.Second):
	}
}

func (x *isolationRun) sampleLoop() {
	defer close(x.convStopped)
	x.takeSample()
	tick := time.NewTicker(250 * time.Millisecond)
	defer tick.Stop()
	for {
		select {
		case <-x.ctx.Done():
			return
		case <-x.stopConv:
			x.takeSample()
			return
		case <-tick.C:
			x.takeSample()
		}
	}
}

func leaseCheckOnly(v any) any {
	items, ok := v.([]any)
	if !ok {
		return v
	}
	for _, item := range items {
		m, ok := item.(map[string]any)
		if !ok {
			continue
		}
		if name, _ := m["name"].(string); name == "lease" {
			return m
		}
	}
	return nil
}

// memberGrants runs before any disruption. The only mutation in this window
// is the real control runtime's future-grant eligibility switch.
func (x *isolationRun) memberGrants() {
	c := iso.Check{Name: "provisional lease reports AWAITING_MEMBER_GRANTS", Status: iso.StatusFail, Owner: "w2-agent-core / w3-wire-nodes", Actual: "D-087 transition incomplete; see member-grants-transition.json"}
	evidence := map[string]any{"run_id": x.records["AT10"].RunID, "recipe_executed": false, "historical_isolation_ms": []int{4767, 5021, 5025}, "devstack_ms": []int{3980, 4112}}
	defer func() {
		evidence["status"] = c.Status
		x.extra("member-grants-transition.json", evidence)
		x.add("AT10", c)
	}()
	var last json.RawMessage
	stableSince := time.Now()
	settling := []map[string]any{}
	wait, cancel := context.WithTimeout(x.ctx, 4*time.Minute)
	stable := waitUntil(wait, time.Second, func() bool {
		v, e := x.stack.MemberGrantInvariants(wait, x.r.nodeID)
		if e != nil {
			stableSince = time.Now()
			return false
		}
		if !bytes.Equal(v, last) {
			last = v
			stableSince = time.Now()
			settling = append(settling, map[string]any{"at": time.Now().UTC(), "snapshot": v})
		}
		return time.Since(stableSince) >= 65*time.Second
	})
	cancel()
	evidence["settling"] = settling
	if !stable {
		c.Actual = "D-087 initial invariants did not settle for 65 seconds"
		return
	}
	before, e := x.stack.MemberGrantInvariants(x.ctx, x.r.nodeID)
	evidence["before"] = before
	var inv struct {
		Node        []string          `json:"node"`
		Publication []string          `json:"publication"`
		Frozen      bool              `json:"frozen"`
		Bindings    []json.RawMessage `json:"bindings"`
		Offline     int               `json:"offline_seconds"`
	}
	if e != nil || json.Unmarshal(before, &inv) != nil || len(inv.Node) != 5 || inv.Node[1] != "ready" || inv.Node[2] != "passed" || inv.Node[3] != inv.Node[4] || inv.Frozen || len(inv.Bindings) == 0 || len(inv.Publication) != 2 || inv.Publication[0] != "published" || inv.Publication[1] != "active" || inv.Offline != 60 {
		c.Actual = "D-087 clean fixed-generation baseline absent"
		return
	}
	initial := x.proof(x.a1)
	evidence["initial_counted_forwarding"] = initial
	if !initial.Valid() {
		c.Actual = "D-087 initial native counted forwarding unavailable"
		return
	}
	// Include the real baseline/meter/tunnel results before changing eligibility.
	x.observation(&c, x.a1)
	falseFlip, e := x.stack.SetMemberGrantsReady(x.ctx, false)
	evidence["false_flip"] = falseFlip
	if e != nil {
		c.Actual = "D-087 dedicated control seam unavailable"
		return
	}
	evidence["recipe_executed"] = true
	// Cleanup restores only eligibility after a failed measurement; it cannot
	// supply a qualifying receipt or change a failed result.
	defer func() { _, _ = x.stack.SetMemberGrantsReady(context.WithoutCancel(x.ctx), true) }()
	var events []GrantWireEvent
	var provisional, serving api.ObservationReport
	var provisionalEvent, servingEvent GrantWireEvent
	var emptyEvent, populatedEvent GrantWireEvent
	await := func(after time.Time, want bool) bool {
		wait, cancel := context.WithTimeout(x.ctx, 3*time.Minute)
		defer cancel()
		return waitUntil(wait, time.Second, func() bool {
			var err error
			events, err = x.stack.MemberGrantEvidence(wait)
			if err != nil {
				return false
			}
			for _, v := range events {
				report, ok := qualifyingGrantReport(v, x.r.nodeID, after, want)
				generation, genErr := report.AppliedConfigGeneration.Get()
				if !ok || genErr != nil || fmt.Sprint(generation) != inv.Node[3] {
					continue
				}
				id, _ := report.Lease.LeaseId.Get()
				for _, l := range events {
					if l.Kind != "lease" || l.Status != 200 || !l.Started.After(after) || l.Completed.After(v.Started) {
						continue
					}
					var lease struct {
						Signed struct {
							ID         uuid.UUID         `json:"lease_id"`
							Generation string            `json:"config_generation"`
							Members    []json.RawMessage `json:"members"`
						} `json:"signed"`
						Signatures []json.RawMessage `json:"signatures"`
					}
					if json.Unmarshal(l.Response, &lease) != nil || lease.Signed.ID != id || lease.Signed.Generation != inv.Node[3] || len(lease.Signatures) == 0 || (len(lease.Signed.Members) == 0) != want {
						continue
					}
					if want {
						provisional, provisionalEvent, emptyEvent = report, v, l
					} else {
						serving, servingEvent, populatedEvent = report, v, l
					}
					return true
				}
			}
			return false
		})
	}
	defer func() { x.extra("member-grants-wire.json", events) }()
	provisionalOK := await(falseFlip.Completed, true)
	evidence["provisional_receipt"], evidence["empty_lease"] = provisionalEvent, emptyEvent
	middle, middleErr := x.stack.MemberGrantInvariants(x.ctx, x.r.nodeID)
	evidence["provisional_invariants"] = middle
	if !provisionalOK {
		c.Actual = "D-087 no accepted qualifying report joined to the natural empty lease"
		return
	}
	trueFlip, e := x.stack.SetMemberGrantsReady(x.ctx, true)
	evidence["true_flip"] = trueFlip
	if e != nil {
		c.Actual = "D-087 true flip failed"
		return
	}
	servingOK := await(trueFlip.Completed, false)
	evidence["serving_receipt"], evidence["populated_lease"] = servingEvent, populatedEvent
	if servingOK {
		c.Positive = x.proof(x.a1)
	}
	after, afterErr := x.stack.MemberGrantInvariants(x.ctx, x.r.nodeID)
	evidence["after"], evidence["counted_forwarding"] = after, c.Positive
	events, e = x.stack.MemberGrantEvidence(x.ctx)
	same := middleErr == nil && afterErr == nil && bytes.Equal(before, middle) && bytes.Equal(before, after)
	emptyID, _ := provisional.Lease.LeaseId.Get()
	fullID, _ := serving.Lease.LeaseId.Get()
	distinct := provisional.ReportId != serving.ReportId && emptyID != fullID
	hashes, wireOK := 0, e == nil
	for _, v := range events {
		for _, pair := range [][2]string{{v.RequestWire, v.RequestSHA256}, {v.ResponseWire, v.ResponseSHA256}} {
			if pair[0] != "" {
				sum := sha256.Sum256([]byte(pair[0]))
				hashes++
				wireOK = wireOK && hex.EncodeToString(sum[:]) == pair[1]
			}
		}
	}
	duration := servingEvent.Completed.Sub(trueFlip.Completed).Milliseconds()
	evidence["proof"] = map[string]any{"invariants_byte_equal": same, "distinct_reports_and_leases": distinct, "empty_lease_report_join": provisionalOK, "populated_lease_report_join": servingOK, "wire_hashes": hashes, "wire_hashes_valid": wireOK, "true_to_serving_receipt_ms": duration, "generation": inv.Node[3]}
	if same && distinct && wireOK && servingOK && c.Positive.Valid() {
		c.Status = iso.StatusPass
		c.Owner = ""
		c.Actual = fmt.Sprintf("D-087 real false/true future-grant transition; accepted reports joined to empty/populated signed leases; fixed generation %s and byte-identical invariants; true to accepted serving %d ms; independent native admission counter increased", inv.Node[3], duration)
	}
}

// TestIsolationRecoverySampling observes the already-running assigned stack.
// It performs no lifecycle, API, nft or tc mutation. The assembled scenario
// supplies its ordinary restart; this observer retains a real provisional
// recovery window if that product transition occurs.
func TestIsolationRecoverySampling(t *testing.T) {
	if os.Getenv("FVPN_E2E") != "1" || os.Getenv("FVPN_E2E_PREFIX") != "w4-isolation" || os.Getenv("FVPN_E2E_PORT_BASE") != "36000" {
		t.Skip("requires the assigned running assembled stack")
	}
	ctx, cancel := context.WithTimeout(t.Context(), 40*time.Minute)
	defer cancel()
	lab, err := openLab()
	if err != nil {
		t.Fatal(err)
	}
	defer lab.transport.CloseIdleConnections()
	dir := filepath.Join(moduleRoot(), "docs/e2e/at/isolation")
	raw, err := os.ReadFile(filepath.Join(dir, "AT10.json"))
	if err != nil {
		t.Fatal(err)
	}
	var rec iso.Record
	if err := json.Unmarshal(raw, &rec); err != nil {
		t.Fatal(err)
	}
	var nodeID uuid.UUID
	for _, o := range rec.Observations {
		if o.Name == "observation_report" {
			var report struct {
				NodeID uuid.UUID `json:"node_id"`
			}
			if json.Unmarshal([]byte(o.Detail), &report) == nil {
				nodeID = report.NodeID
			}
		}
	}
	if nodeID == uuid.Nil {
		t.Fatal("no node identity in this run's serving observation")
	}
	x := &isolationRun{t: t, ctx: ctx, r: &runner{t: t, lab: lab, nodeID: nodeID}}
	tick := time.NewTicker(250 * time.Millisecond)
	defer tick.Stop()
	seen := false
	admittedAt := ""
	totalSamples := 0
	for ctx.Err() == nil {
		x.takeSample()
		totalSamples++
		if !seen && len(x.conv) > 240 {
			x.conv = append([]map[string]any(nil), x.conv[len(x.conv)-120:]...)
		}
		row := x.conv[len(x.conv)-1]
		if c, ok := row["lease_check"].(map[string]any); ok && c["error_code"] == "AWAITING_MEMBER_GRANTS" {
			seen = true
		}
		if seen && row["forwarding"] == true {
			if n, ok := row["forward_member_accept"].(int); ok && n > 0 {
				if result := iso.EvaluateProvisional(x.conv); result.Status == iso.StatusPass && row["lease_generation"] == result.Generation {
					admittedAt, _ = row["at"].(string)
					break
				}
			}
		}
		select {
		case <-ctx.Done():
		case <-tick.C:
		}
	}
	build, _ := exec.Command("git", "rev-parse", "HEAD").Output()

	doc := map[string]any{"run_id": rec.RunID, "image_build": rec.Build, "sampler_build": strings.TrimSpace(string(build)), "read_only": true, "sample_cadence_ms": 250, "total_samples": totalSamples, "samples": x.conv, "result": result, "forwarding_admitted_at": admittedAt, "positive_reference": "AT13 node restart with durable identity in the same assembled run"}
	body, err := json.MarshalIndent(doc, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := iso.WriteExtra(dir, "provisional-recovery.json", string(body)+"\n"); err != nil {
		t.Fatal(err)
	}
	if result.Status != iso.StatusPass || admittedAt == "" {
		t.Fatalf("recovery window: %+v", result)
	}
	t.Logf("recovery provisional samples=%d ready_to_populated_ms=%d admitted_at=%s", len(result.SampleTimes), result.ReadyToPopulatedMS, admittedAt)
}

func (x *isolationRun) archiveReceiver() {
	if x.r == nil {
		return
	}
	// Quiesce only after all measurements; keep the container available for the
	// shared stream exporter, then let the receiver finish its 100ms drain.
	closure := []map[string]any{}
	for _, container := range []string{x.r.lab.node, x.r.lab.prefix + "-node-clone"} {
		start := time.Now()
		out, err := exec.Command("docker", "exec", container, "agent-service.sh", "stop").CombinedOutput()
		closure = append(closure, map[string]any{"container": container, "start": start.UTC().Format(time.RFC3339Nano), "end": time.Now().UTC().Format(time.RFC3339Nano), "ok": err == nil, "output": iso.Redact(string(out))})
	}
	time.Sleep(300 * time.Millisecond)
	x.extra("receiver-closure.json", closure)
	// Producer measurement is a detached sibling of D-106. Export every line
	// after stopping the source; never silently discard a failed/partial record.
	for _, item := range []struct{ source, name string }{
		{"isolation-operations.jsonl", "producer-operations.jsonl"},
		{"isolation-clock.json", "producer-clock.json"},
	} {
		path := filepath.Join(x.r.lab.root, ".agent", "tmp", item.name)
		_, err := x.stack.ExportContainerFile(x.ctx, "node", "/var/lib/vpn-node/e2e/"+item.source, path)
		if err != nil {
			x.t.Errorf("producer export %s: %v", item.name, err)
			continue
		}
		raw, err := os.ReadFile(path)
		if err == nil {
			err = iso.WriteExtra(x.dir, item.name, string(raw))
		}
		if err != nil {
			x.t.Errorf("producer retention %s: %v", item.name, err)
		}
	}
	x.retainReasonLogs("final")
	for _, name := range []string{"agent", "watchdog"} {
		var all bytes.Buffer
		for _, label := range []string{"before-restart", "before-clone", "final"} {
			raw, err := os.ReadFile(filepath.Join(x.dir, name+"-"+label+".log.txt"))
			if err == nil {
				all.Write(raw)
			}
		}
		_ = iso.WriteExtra(x.dir, name+".log.txt", all.String())
	}

	// Stream the full private capture to task disk, outside diagnostic stdout.
	scratch := filepath.Join(x.r.lab.root, ".agent", "tmp", "isolation-receiver.jsonl")
	_, err := (&Stack{lab: x.r.lab}).ExportContainerFile(x.ctx, "node", "/var/lib/vpn-node/e2e/isolation-observations.jsonl", scratch)
	nftRaw, _ := exec.Command("docker", "exec", x.r.lab.node, "cat", "/var/lib/vpn-node/e2e/isolation-nft-events.jsonl").Output()
	nftRows := []json.RawMessage{}
	for _, line := range bytes.Split(nftRaw, []byte("\n")) {
		if json.Valid(line) {
			nftRows = append(nftRows, append(json.RawMessage(nil), line...))
		}
	}
	x.extra("nft-event-receipts.json", map[string]any{"clock": "CLOCK_MONOTONIC; passive netlink userspace receipt; not render/command timestamps", "events": nftRows})
	if err != nil {

		x.extra("receiver-capture.json", map[string]any{"error": "capture export failed"})
		return
	}
	raw, err := os.ReadFile(scratch)
	if err != nil {
		x.extra("receiver-capture.json", map[string]any{"error": "capture unavailable"})
		return
	}
	var rows []json.RawMessage
	for _, line := range bytes.Split(raw, []byte("\n")) {
		if len(line) > 0 && json.Valid(line) {
			rows = append(rows, append(json.RawMessage(nil), line...))
		}
	}
	x.extra("receiver-capture.json", map[string]any{"run_id": x.records["AT10"].RunID, "build": x.records["AT10"].Build, "drain_ms": 100, "heartbeat_seconds": 1, "poll_seconds": 1, "observation_seconds": 1, "scope": "Transparent loopback mTLS control receiver, every observation POST body and response plus renewal transactions; no headers. Installed before enrollment and retained across source restarts.", "transactions": rows})
	conn, err := x.r.lab.db(x.ctx)
	if err != nil {
		return
	}
	defer conn.Close(x.ctx)
	var history []byte
	if conn.QueryRow(x.ctx, `SELECT jsonb_build_object('accepted_report_ids', (SELECT jsonb_agg(report_id) FROM node_observations WHERE node_id=$1), 'leases', (SELECT jsonb_agg(jsonb_build_object('id',id,'issued_at',issued_at,'envelope',envelope) ORDER BY issued_at) FROM serving_leases WHERE node_id=$1),'desired', (SELECT envelope FROM desired_state_envelopes WHERE node_id=$1))`, x.r.nodeID).Scan(&history) == nil {
		x.extra("lease-history.json", json.RawMessage(history))
	}
}

type isolationCapture struct {
	cmd  *exec.Cmd
	out  bytes.Buffer
	name string
}

func (x *isolationRun) continuousCapture(container, iface, filter, name string, seconds int) *isolationCapture {
	c := &isolationCapture{name: name}
	c.cmd = exec.CommandContext(x.ctx, "docker", "exec", container, "timeout", "-s", "INT", strconv.Itoa(seconds), "tcpdump", "-n", "-tt", "--time-stamp-precision=nano", "-l", "-i", iface, filter)
	c.cmd.Stdout = &c.out
	c.cmd.Stderr = &c.out
	if err := c.cmd.Start(); err != nil {
		x.t.Logf("capture start: %v", err)
	}
	return c
}
func (x *isolationRun) finishCapture(c *isolationCapture) {
	if c == nil {
		return
	}
	_ = c.cmd.Wait()
	_ = iso.WriteExtra(x.dir, c.name, c.out.String())
}
func (x *isolationRun) startContinuousExpiry(held leaseSnap) func() iso.DeadlineVerdict {
	filter := "host " + x.a1.originIP() + " and udp port 8080"
	cap := x.continuousCapture(x.r.lab.node, "any", filter, "whole-node-node-capture.txt", 25)
	_ = exec.Command("docker", "exec", "-d", x.r.lab.node, "/tmp/isolation-probe", "packet-clock", "24", "/tmp/isolation-packets.jsonl").Run()
	time.Sleep(100 * time.Millisecond)

	stop, done := make(chan struct{}), make(chan struct{})
	go func() {
		defer close(done)
		var rows []map[string]any
		defer func() { x.extra("whole-node-continuous-reads.json", map[string]any{"lease": held, "reads": rows}) }()
		for {
			select {
			case <-stop:
				return
			case <-x.ctx.Done():
				return
			default:
			}
			start := time.Now()
			raw, _ := x.r.lab.docker(x.r.lab.node, "sh", "-c", "/tmp/isolation-probe clock; nft list set inet fvpn allowed_v4; nft list chain inet fvpn forward; cat /run/vpn-node/heartbeat; /tmp/isolation-probe clock")
			rows = append(rows, map[string]any{"read_start": start.UTC().Format(time.RFC3339Nano), "read_end": time.Now().UTC().Format(time.RFC3339Nano), "readback": string(raw)})
			time.Sleep(100 * time.Millisecond)
		}
	}()
	return func() iso.DeadlineVerdict {
		x.finishCapture(cap)
		close(stop)
		<-done
		read := func(container, path string) []byte {
			raw, _ := exec.Command("docker", "exec", container, "cat", path).Output()
			rows := []json.RawMessage{}
			for _, line := range bytes.Split(raw, []byte("\n")) {
				if json.Valid(line) {
					rows = append(rows, append(json.RawMessage(nil), line...))
				}
			}
			out, _ := json.Marshal(rows)
			return out
		}
		probes := read(x.a1.lab.member, "/tmp/isolation-expiry-timed.jsonl")
		packets := read(x.r.lab.node, "/tmp/isolation-packets.jsonl")
		x.extra("whole-node-probes.json", map[string]any{"lease": held, "probes": json.RawMessage(probes)})
		x.extra("whole-node-packets.json", json.RawMessage(packets))
		return iso.EvaluateDeadline(probes, packets, held.DeadlineMono)
	}

	fixture, err := x.stack.StartClonedIdentityFixture(x.ctx, "node")
	evidence := map[string]any{"started_at": at.UTC().Format(time.RFC3339Nano), "owner": "ENR-02 / ENR-03; this is the consumer latch observation only"}
	defer func() { x.extra("identity-freeze.json", evidence) }()
	if err != nil {
		evidence["status"] = "blocked"
		evidence["detail"] = err.Error()
		return
	}
	var rows []map[string]any
	for range 90 {
		obs, e := fixture.Observe(x.ctx)
		row := x.networkSnapshot("clone acceptance window")
		row["clone_observation"] = obs
		if e != nil {
			row["clone_read_error"] = e.Error()
		}
		rows = append(rows, row)
		if obs.MappingFrozen && strings.Contains(row["readback"].(string), "identity_frozen") {
			time.Sleep(3 * time.Second)
			rows = append(rows, x.networkSnapshot("freeze latch held"))
			break
		}
		time.Sleep(time.Second)
	}
	evidence["rows"] = rows
	evidence["status"] = "requires receiver correlation"
	evidence["fixture"] = "same enrolled incarnation presented from a second lab network; not a newly enrolled identity"
}

func (x *isolationRun) reviewFollowup8() {
	cmd := exec.Command("python3", filepath.Join(moduleRoot(), "tests/e2e/at/isolation/review_followup8.py"), x.dir)
	if out, err := cmd.CombinedOutput(); err != nil {
		x.t.Logf("review: %v %s", err, out)
		return
	}
	raw, err := os.ReadFile(filepath.Join(x.dir, "recovery-verdict.json"))
	if err != nil {
		return
	}
	var v struct {
		Passed bool   `json:"passed"`
		Status string `json:"status"`
	}
	if json.Unmarshal(raw, &v) != nil {
		return
	}
	for i, c := range x.records["AT10"].Checks {
		if c.Name == "foreign ingress redirect containment and D-096 recovery" {
			c.Actual += "; six assertions: " + string(raw)
			if v.Passed {
				c.Status = iso.StatusPass
				c.Owner = ""
			} else if c.Status != iso.StatusFail {
				c.Status = iso.StatusBlocked
			}
			x.records["AT10"].Checks[i] = c
		}
	}
}

// Read-only snapshot of state that production persists only after verified Apply.
func (x *isolationRun) captureApplied() {
	dir := x.t.TempDir()
	for _, name := range []string{"agent.db", "agent.db-wal", "agent.db-shm"} {
		_ = exec.Command("docker", "cp", x.r.lab.node+":/var/lib/vpn-node/state/"+name, filepath.Join(dir, name)).Run()
	}
	db, err := sql.Open("sqlite", "file:"+filepath.Join(dir, "agent.db")+"?mode=ro")
	if err != nil {
		return
	}
	defer db.Close()
	result := map[string]any{"read_at": time.Now().UTC().Format(time.RFC3339Nano), "source": "read-only node SQLite snapshot; production persistence after verified Apply"}
	for _, key := range []string{"retained_applied", "member_path"} {
		var raw string
		if db.QueryRow(`SELECT value FROM meta WHERE key=?`, key).Scan(&raw) == nil && json.Valid([]byte(raw)) {
			result[key] = json.RawMessage(raw)
		}
	}
	var applied int64
	_ = db.QueryRow(`SELECT applied_generation FROM desired WHERE id=1`).Scan(&applied)
	result["applied_generation"] = fmt.Sprint(applied)
	lease, ok := x.readNodeLease()
	if ok {
		result["lease"] = lease
	}
	x.extra("recovery-applied-state.json", result)
}
