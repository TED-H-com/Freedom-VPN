package main

import (
 "errors"
 "fmt"
 "os"
 "github.com/TED-H-com/Freedom-VPN/node/enforce"
)

func main() {
 for _, path := range os.Args[1:] {
  raw, err := os.ReadFile(path)
  if err != nil { panic(err) }
  _, err = enforce.ParseState(raw)
  var duplicate *enforce.DuplicateChainError
  if err != nil {
   fmt.Printf("%s rejected: duplicate_kind=%t\n", path, errors.As(err, &duplicate))
   os.Exit(1)
  }
  fmt.Printf("%s: ParseState passed\n", path)
 }
}
